def load_data():
    try:
        sh = gc.open_by_key(SHEET_ID)
        ws = sh.sheet1
        data = ws.get_all_records()
        if not data:
            return pd.DataFrame(columns=EXPECTED_COLS)
        df = pd.DataFrame(data)
        df = df.reindex(columns=EXPECTED_COLS, fill_value="-")
        return df
    except Exception as e:
        st.error(f"Gagal membaca Google Sheets: {e}")
        return pd.DataFrame(columns=EXPECTED_COLS)

def save_data(df):
    try:
        sh = gc.open_by_key(SHEET_ID)
        ws = sh.sheet1
        df = df.fillna("-").astype(str)
        df.replace(["nan", "None", ""], "-", inplace=True)
        data_to_write = [EXPECTED_COLS] + df[EXPECTED_COLS].values.tolist()
        ws.clear()
        ws.update(values=data_to_write, range_name="A1")
    except Exception as e:
        st.error(f"Gagal menyimpan ke Google Sheets: {e}")

def upload_to_drive(file_obj, filename, folder_id):
    try:
        file_bytes = file_obj.getvalue()
        encoded_file = base64.b64encode(file_bytes).decode('utf-8')
        payload = {"fileData": encoded_file, "filename": filename, "mimeType": file_obj.type, "folderId": folder_id}
        response = requests.post(APPS_SCRIPT_URL, json=payload, timeout=60)
        res_json = response.json()
        if res_json.get("status") == "success":
            return res_json.get("fileId")
        else:
            st.error(f"Gagal upload {filename}: {res_json.get('message')}")
            return None
    except Exception as e:
        st.error(f"Error koneksi upload {filename}: {e}")
        return None

def clean_wa_number(phone_str):
    if not phone_str or pd.isna(phone_str) or str(phone_str).strip() == "-" or str(phone_str).strip() == "":
        return ""
    clean_num = re.sub(r'\D', '', str(phone_str))
    if clean_num.startswith('0'): clean_num = '62' + clean_num[1:]
    elif not clean_num.startswith('62') and clean_num != "": clean_num = '62' + clean_num
    return clean_num
